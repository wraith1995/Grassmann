def dot(x, y):
    return (x * y).sum(-1)

def minors(a, b, pairs):        # a_i b_j - a_j b_i for each (i, j)
    i, j = (list(t) for t in zip(*pairs))
    return a[..., i] * b[..., j] - a[..., j] * b[..., i]

def triples(a, b, c, rows):     # det[a,b,c] on coordinates (i,j,k)
    return stack([dot(a[..., [i, j, k]],
                      minors(b, c, [(j, k), (k, i), (i, j)]))
                  for i, j, k in rows], -1)

# Cauchy-Binet: k! det of the Gram matrix of inner products on V
def inner_of_wedges(V, us, vs):
    pair = G.inner(V).at([0, 1])
    gram = [[pair(u, v) for v in vs] for u in us]
    det = sum(sign(p) * prod(gram[i][p[i]] for i in range(len(us)))
              for p in perms(len(us)))
    return factorial(len(us)) * det

cb = inner_of_wedges(V, [a, b], [c, d])
